    Sparse Matrix Multiplication

    Verilog implementations for various matrix multiplication algorithms targeting RISC-V vector processors, based on the paper Optimizing Structured-Sparse Matrix Multiplication in RISC-V Vector Processors.

    Applicaton

    Xilinx Vivado 2024.2

    Algorithm 1

    Row-wise vector matrix multiplication with data placed only in the vector register file

    set vector length
for i=0 until i=num_of_rows_of_A-1 do
    vload v0, A[i,:]                  ▷ load the ith row of A
    v2 = {0,0,...,0}                  ▷ Initialize v2 to zero
    for j=0 until j=num_of_col_of_A-1 do
        vload v1, B[j,:]              ▷ load jth row of B
        s0 = A[i,0]                   ▷ move A[i,0] to a scalar reg
        v2 += s0 * v1                 ▷ scalar-vector mul-acc
        v0 = v0 >> 1                  ▷ vector 1 slide to the right
    end for
    vstore v2, C[i,:]                 ▷ store to mem the ith row of C
end for
    algorithm1_des.v
    module algorithm1_des #(parameter VECTOR_LEN = 8,
                        parameter ROW_A = 3,       
                        parameter COL_A = 3,  
                        parameter ROW_B = 3,       
                        parameter COL_B = 4)
                        (input clk,
                        input rstn,
                        input [(ROW_A * COL_A * VECTOR_LEN) - 1:0] A,
                        input [(ROW_B * COL_B * VECTOR_LEN) - 1:0] B,
                        output reg [(ROW_A * COL_B * VECTOR_LEN) - 1:0]C);



//register array declarations              
reg [(COL_A * VECTOR_LEN)-1:0] v0;
reg [(COL_B * VECTOR_LEN)-1:0] v1;
reg [(COL_B * VECTOR_LEN)-1:0] v2;

integer i, j, k; 
reg [VECTOR_LEN-1:0] s;

//matrix multiplication
always@(posedge clk or negedge rstn)
begin
    if (!rstn)
    begin
        //Initialize all the vector register elements to zero.
        v0 <= 'd0;
        v1 <= 'd0;
        v2 <= 'd0;


    end
    //matrix multiplication
    else
    begin
        for (i = 0; i < ROW_A; i = i + 1)
        begin
                v0 = A[i * COL_A * VECTOR_LEN +: (COL_A * VECTOR_LEN)];
                v2 = 'd0;

                for (j = 0; j < COL_A; j = j + 1)
                begin
                    v1 = B[j * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)];
                    s = v0[VECTOR_LEN-1:0];  

                    //multiply accumulate                        
                    for (k = 0; k < COL_B; k = k + 1)
                        v2[k*VECTOR_LEN +: VECTOR_LEN] = v2[k*VECTOR_LEN +: VECTOR_LEN] + s * v1[k*VECTOR_LEN +: VECTOR_LEN];

                    v0 = v0 >> VECTOR_LEN; 
                end   

                C[i * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)] = v2;
        end
    end           
end
endmodule

    Algorithm 3

    Row-wise vector matrix multiplication with data placed only in the VRF by utilizing the vrgather instruction

    set vector length
for i=0 until i=num_of_rows_of_A-1 do
    vload v0, A[i,:]                  ▷ load ith row of A
    v3 = {0,0,...,0}                  ▷ Initialize v3 to zero
    for j=0 until j=num_of_col_of_A-1 do
        vload v1, B[j,:]              ▷ load jth row of B
        vrgather.vx v2, v0, j         ▷ fill v2 with the jth el. of A
        v3 += v2 * v1                 ▷ vector-vector mul-acc
    end for
    vstore v3, C[i,:]                 ▷ store to mem the ith row of C
end for

    Algorithm 1s

    Row-wise vector sparse-dense matrix multiplication with data placed only in the vector register file

    set vector length
for i=0 until i=num_of_rows_of_A-1 do
    vload v0, values[i,:]             ▷ load the non-zero val. of A
    v2 = {0,0,...,0}                  ▷ Initialize v2 to zero
    for j=0 until j=(num_of_col_of_A/M)*N-1 do
        block_id = ⌊j/N⌋
        load s1, col_idx[i,j]         ▷ load a column index
        s1 += block_id * M             ▷ calculate actual col. idx
        vload v1, B[s1,:]             ▷ load the selected row of B
        s0 = A[i,0]                  ▷ move A[i,0] to a scalar reg
        v2 += s0 * v1                 ▷ scalar-vector mul-acc
        v0 = v0 >> 1                 ▷ vector 1 slide to the right
    end for
    vstore v2, C[i,:]                 ▷ store to mem the ith row of C
end for
    algorithm1s_des
    module Algorithm1s_des #(parameter VECTOR_LEN = 8,
                    parameter ROW_A = 3,       
                    parameter COL_A = 3,  
                    parameter ROW_B = 3,       
                    parameter COL_B = 4,
                    parameter M = 3,
                    parameter N = 2)
                (input clk,
                    input rstn,
                    input [(ROW_A * ((COL_A / M) * N) * VECTOR_LEN) - 1:0] val,     // non-zero values
                    input [(ROW_A * ((COL_A / M) * N) * VECTOR_LEN) - 1:0] col_idx, // column indices of non-zero values
                    input [(ROW_B * COL_B * VECTOR_LEN) - 1:0] B,
                    output reg [(ROW_A * COL_B * VECTOR_LEN) - 1:0] C);

// number of non-zero entries per row
localparam no_of_non_zero_val = ((COL_A / M) * N);             

// unpacked col index storage
reg [VECTOR_LEN-1 : 0] col_idx_matrix [0: ROW_A-1][0: no_of_non_zero_val-1];  

integer i_c, j_c;

// expand packed col_idx input into 2D array
always @(*) begin
    for(i_c = 0; i_c < ROW_A ; i_c = i_c + 1) begin
        for (j_c = 0; j_c < no_of_non_zero_val ; j_c = j_c + 1) begin
            col_idx_matrix[i_c][j_c] = col_idx[(i_c * no_of_non_zero_val + j_c) * VECTOR_LEN +: VECTOR_LEN];
        end
    end
end

// register array declarations              
reg [(no_of_non_zero_val * VECTOR_LEN)-1:0] v0;
reg [(COL_B * VECTOR_LEN)-1:0] v1;
reg [(COL_B * VECTOR_LEN)-1:0] v2;

integer i, j, k; 
reg [VECTOR_LEN-1:0] s0;
reg [VECTOR_LEN-1:0] s1;
reg [VECTOR_LEN-1:0] block_id;

// matrix multiplication
always@(posedge clk or negedge rstn) begin
    if (!rstn) begin
        v0 <= 'd0;
        v1 <= 'd0;
        v2 <= 'd0; 
        // C  <= 'd0;
    end else begin
        for (i = 0; i < ROW_A; i = i + 1) begin
            v0 = val[i * no_of_non_zero_val * VECTOR_LEN +: (no_of_non_zero_val * VECTOR_LEN)];
            v2 = 'd0;

            for (j = 0; j < no_of_non_zero_val; j = j + 1) begin
                block_id = j / N;
                s1 = col_idx_matrix[i][j];
                s1 = s1 + (block_id * M); 
                v1 = B[s1 * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)];
                s0 = v0[VECTOR_LEN-1:0];   
                // multiply accumulate                        
                for (k = 0; k < COL_B; k = k + 1)
                    v2[k*VECTOR_LEN +: VECTOR_LEN] = v2[k*VECTOR_LEN +: VECTOR_LEN] + (s0 * v1[k*VECTOR_LEN +: VECTOR_LEN]);

                v0 = v0 >> VECTOR_LEN;
            end   

            C[i * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)] = v2;
        end
    end           
end

endmodule

    module Algorithm6_des#(
    parameter VECTOR_LEN = 8,
    parameter ROW_A = 3,       
    parameter COL_A = 3,  
    parameter ROW_B = 3,       
    parameter COL_B = 4,
    parameter M = 3,
    parameter N = 2,
    parameter L = 3,
    parameter VL = 3
)(
    input clk,
    input rstn,
    input [(ROW_A * ((COL_A / M) * N) * VECTOR_LEN) - 1:0] val,     // non-zero values
    input [(ROW_A * ((COL_A / M) * N) * VECTOR_LEN) - 1:0] col_idx, // column indices of non-zero values
    input [(ROW_B * COL_B * VECTOR_LEN) - 1:0] B,
    output reg [(ROW_A * COL_B * VECTOR_LEN) - 1:0] C
);

// number of non-zero entries per row
localparam no_of_non_zero_val = ((COL_A / M) * N);            

// unpacked col index storage
reg [VECTOR_LEN-1 : 0] col_idx_matrix [0: ROW_A-1][0: no_of_non_zero_val-1];  

integer i_c, j_c;

// expand packed col_idx into 2D array
always @(*) 
begin
    for(i_c = 0; i_c < ROW_A ; i_c = i_c + 1) 
    begin
        for (j_c = 0; j_c < no_of_non_zero_val ; j_c = j_c + 1) 
        begin
            col_idx_matrix[i_c][j_c] = col_idx[(i_c * no_of_non_zero_val + j_c) * VECTOR_LEN +: VECTOR_LEN];
        end
    end    
end

// register arrays              
reg [(no_of_non_zero_val * VECTOR_LEN)-1:0] vs2;
reg [(COL_B * VECTOR_LEN)-1:0] v[L-1:0];
reg [(COL_B * VECTOR_LEN)-1:0] vs1;
reg [(COL_B * VECTOR_LEN)-1:0] vd;

integer i, j, k, l, m; 
reg [VECTOR_LEN-1:0] s1;
reg [VECTOR_LEN-1:0] block_id;

// matrix multiplication
always@(posedge clk or negedge rstn) 
begin
    if (!rstn) 
    begin
        vs2 <= 'd0;
        vs1 <= 'd0;
        vd  <= 'd0;
        for(l = 0; l < L; l = l + 1)
            v[l] <= 'd0;    // FIX: use l instead of k
        C <= 'd0;
    end 
    else 
    begin
        for (i = 0; i < ROW_A; i = i + 1) 
        begin
            vs2 = val[i * no_of_non_zero_val * VECTOR_LEN +: (no_of_non_zero_val * VECTOR_LEN)];
            vd  = C[i * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)];

            for (m = 0; m < (M * VL / (N * L)); m = m + 1) 
            begin   
                for (k = (L * m); k < (m + 1)*L ; k = k + 1)
                begin
                    v[k] <= B[k * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)];
                end 
                for (j = (L * N * m)/M ; j < ((L * N * (m + 1))/M); j = j + 1 )
                begin
                    block_id = j / N;
                    s1 = col_idx_matrix[i][j];
                    s1 = s1 + (block_id * M); 

                    // fetch row of B
                    vs1 = v[s1];

                    // multiply accumulate                        
                    for (k = 0; k < COL_B; k = k + 1)
                        vd[k*VECTOR_LEN +: VECTOR_LEN] = vd[k*VECTOR_LEN +: VECTOR_LEN] + (vs2[VECTOR_LEN-1 : 0] * vs1[k*VECTOR_LEN +: VECTOR_LEN]);        

                vs2 = vs2 >> VECTOR_LEN;                    
                end

            end   

            C[i * COL_B * VECTOR_LEN +: (COL_B * VECTOR_LEN)] <= vd;
        end
    end           
end
